### 2. Create app/api/admin/imports/[id]/ai/chat/route.ts

```typescript
// POST /api/admin/imports/:id/ai/chat
// Body: { message: string, history: ChatMessage[] }
// Returns: Server-Sent Events stream
// Each SSE event: { type: 'chunk', text: '...' } or { type: 'actions', actions: [...] } or { type: 'done' }

export async function POST(req: Request, { params }: { params: { id: string } }) {
  const { message, history } = await req.json();
  const importJobId = params.id;

  // Fetch job context from DB
  const supabase = createAdminClient();
  const { data: job } = await supabase
    .from('import_jobs')
    .select('*')
    .eq('id', importJobId)
    .single();

  // Fetch recent errors summary
  const { data: recentErrors } = await supabase
    .from('staging_import_contributions')
    .select('validation_errors')
    .eq('import_job_id', importJobId)
    .eq('validation_status', 'invalid')
    .limit(100);

  // Build error summary
  const errorSummary = buildErrorSummary(recentErrors ?? []);

  // Stream response
  const encoder = new TextEncoder();
  const stream = new ReadableStream({
    async start(controller) {
      try {
        const { fullResponse, actions } = await streamMigrationChat(
          {
            importJobId,
            message,
            history,
            jobContext: {
              status: job?.status ?? 'unknown',
              recordsExtracted: job?.total_records_extracted ?? 0,
              recordsLoaded: job?.records_loaded ?? 0,
              recordsFailed: job?.records_failed ?? 0,
              recentErrors: errorSummary,
              reconciliation: job?.reconciliation_data,
            },
          },
          (chunk) => {
            controller.enqueue(encoder.encode(`data: ${JSON.stringify({ type: 'chunk', text: chunk })}\n\n`));
          }
        );

        // Send actions
        if (actions.length > 0) {
          controller.enqueue(encoder.encode(`data: ${JSON.stringify({ type: 'actions', actions })}\n\n`));
        }

        controller.enqueue(encoder.encode(`data: ${JSON.stringify({ type: 'done' })}\n\n`));
        controller.close();
      } catch (err) {
        const message = err instanceof Error ? err.message : 'Unknown error';
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({ type: 'error', message })}\n\n`));
        controller.close();
      }
    },
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive',
    },
  });
}
```
