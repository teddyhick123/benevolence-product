```typescript
import { jsPDF } from 'jspdf';
import autoTable from 'jspdf-autotable';

// Brand colors
export const COLORS = {
  primary: [79, 70, 229] as [number, number, number],      // indigo-600
  primaryLight: [238, 242, 255] as [number, number, number], // indigo-50
  success: [22, 163, 74] as [number, number, number],       // green-600
  warning: [202, 138, 4] as [number, number, number],       // yellow-600
  danger: [220, 38, 38] as [number, number, number],        // red-600
  dark: [17, 24, 39] as [number, number, number],           // gray-900
  muted: [107, 114, 128] as [number, number, number],       // gray-500
  border: [229, 231, 235] as [number, number, number],      // gray-200
  white: [255, 255, 255] as [number, number, number],
};

export function createDoc(): jsPDF {
  return new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });
}

// Page dimensions
export const PAGE = { width: 210, height: 297, margin: 20 };
export const CONTENT_WIDTH = PAGE.width - PAGE.margin * 2;

// Add branded header to a page
export function addPageHeader(doc: jsPDF, title: string, subtitle?: string, pageNum?: number): number {
  // Indigo header bar
  doc.setFillColor(...COLORS.primary);
  doc.rect(0, 0, PAGE.width, 18, 'F');

  // Logo text (replace with image if available)
  doc.setTextColor(...COLORS.white);
  doc.setFontSize(11);
  doc.setFont('helvetica', 'bold');
  doc.text('Benevolence', PAGE.margin, 11);

  // Page number
  if (pageNum !== undefined) {
    doc.setFontSize(9);
    doc.setFont('helvetica', 'normal');
    doc.text(`Page ${pageNum}`, PAGE.width - PAGE.margin, 11, { align: 'right' });
  }

  // Title block
  doc.setTextColor(...COLORS.dark);
  doc.setFontSize(18);
  doc.setFont('helvetica', 'bold');
  doc.text(title, PAGE.margin, 32);

  let y = 38;
  if (subtitle) {
    doc.setFontSize(10);
    doc.setFont('helvetica', 'normal');
    doc.setTextColor(...COLORS.muted);
    doc.text(subtitle, PAGE.margin, y);
    y += 6;
  }

  // Divider
  doc.setDrawColor(...COLORS.border);
  doc.setLineWidth(0.3);
  doc.line(PAGE.margin, y, PAGE.width - PAGE.margin, y);

  return y + 6; // return Y position after header
}

// Add footer to each page
export function addPageFooter(doc: jsPDF, text: string): void {
  const pages = doc.getNumberOfPages();
  for (let i = 1; i <= pages; i++) {
    doc.setPage(i);
    doc.setFontSize(8);
    doc.setTextColor(...COLORS.muted);
    doc.setFont('helvetica', 'normal');
    doc.text(text, PAGE.margin, PAGE.height - 8);
    doc.text(`${i} / ${pages}`, PAGE.width - PAGE.margin, PAGE.height - 8, { align: 'right' });
    // Bottom border
    doc.setDrawColor(...COLORS.border);
    doc.setLineWidth(0.3);
    doc.line(PAGE.margin, PAGE.height - 12, PAGE.width - PAGE.margin, PAGE.height - 12);
  }
}

// Stat card: label + big value
export function addStatCard(
  doc: jsPDF,
  x: number,
  y: number,
  width: number,
  height: number,
  label: string,
  value: string,
  color?: [number, number, number]
): void {
  // Card background
  doc.setFillColor(...COLORS.primaryLight);
  doc.roundedRect(x, y, width, height, 2, 2, 'F');
  // Label
  doc.setFontSize(8);
  doc.setFont('helvetica', 'normal');
  doc.setTextColor(...COLORS.muted);
  doc.text(label.toUpperCase(), x + 4, y + 6);
  // Value
  doc.setFontSize(16);
  doc.setFont('helvetica', 'bold');
  doc.setTextColor(...(color ?? COLORS.dark));
  doc.text(value, x + 4, y + 15);
}

// Section heading
export function addSectionHeading(doc: jsPDF, text: string, y: number): number {
  doc.setFontSize(12);
  doc.setFont('helvetica', 'bold');
  doc.setTextColor(...COLORS.primary);
  doc.text(text, PAGE.margin, y);
  doc.setDrawColor(...COLORS.primary);
  doc.setLineWidth(0.5);
  doc.line(PAGE.margin, y + 1, PAGE.margin + 40, y + 1);
  return y + 7;
}

// Body text
export function addBodyText(doc: jsPDF, text: string, y: number, maxWidth?: number): number {
  doc.setFontSize(10);
  doc.setFont('helvetica', 'normal');
  doc.setTextColor(...COLORS.dark);
  const lines = doc.splitTextToSize(text, maxWidth ?? CONTENT_WIDTH);
  doc.text(lines, PAGE.margin, y);
  return y + lines.length * 5;
}

// Format currency
export function formatCurrency(amount: number): string {
  return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0 }).format(amount);
}

// Format percentage
export function formatPercent(value: number, decimals = 1): string {
  return `${value.toFixed(decimals)}%`;
}

// Check if we need a new page and add one
export function checkPageBreak(doc: jsPDF, y: number, needed = 20): number {
  if (y + needed > PAGE.height - 20) {
    doc.addPage();
    return 25;
  }
  return y;
}
```

```typescript
import autoTable from 'jspdf-autotable';
import {
  createDoc, addPageHeader, addPageFooter, addStatCard,

  formatCurrency, formatPercent, COLORS, PAGE, CONTENT_WIDTH
} from './pdf-generator';

export interface MigrationReportData {
  jobName: string;
  portfolioName: string;
  sourceSystem: string;
  completedAt: string;
  entityCounts: {
    investees: { loaded: number; failed: number; total: number };
    holdings: { loaded: number; failed: number; total: number };
    contributions: { loaded: number; failed: number; total: number };
    metrics: { loaded: number; failed: number; total: number };
  };
  financialReconciliation: {
    sourceTotal: number;
    loadedTotal: number;
    deltaPercent: number;
  };
  errorsFixed: number;
  aiSuggestionsApplied: number;
  actionItems: string[];
  enrichmentStats: {
    charitiesMatched: number;
    taxYearsDerived: number;
    deductibleAmountsCalculated: number;
  };
  performanceMs?: number;
}

export function generateMigrationReportPDF(data: MigrationReportData): Buffer {
  const doc = createDoc();
  const generatedDate = new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' });

  // Page 1: Executive Summary
  let y = addPageHeader(doc, 'Migration Report', `${data.portfolioName} · ${generatedDate}`, 1);

  // Executive summary box
  doc.setFillColor(238, 242, 255);
  doc.roundedRect(PAGE.margin, y, CONTENT_WIDTH, 24, 2, 2, 'F');
  doc.setFontSize(10);
  doc.setFont('helvetica', 'italic');
  doc.setTextColor(79, 70, 229);
  const totalLoaded = Object.values(data.entityCounts).reduce((sum, e) => sum + e.loaded, 0);
  const totalRecords = Object.values(data.entityCounts).reduce((sum, e) => sum + e.total, 0);
  const successRate = totalRecords > 0 ? Math.round((totalLoaded / totalRecords) * 100) : 0;
  const execSummary = `Successfully migrated ${totalLoaded.toLocaleString()} of ${totalRecords.toLocaleString()} records (${successRate}% success rate) from ${data.sourceSystem} to Benevolence. Financial reconciliation shows a ${formatPercent(data.financialReconciliation.deltaPercent)} delta. AI resolved ${data.errorsFixed} data quality issues automatically.`;
  const execLines = doc.splitTextToSize(execSummary, CONTENT_WIDTH - 8);
  doc.text(execLines, PAGE.margin + 4, y + 7);
  y += 30;

  // Stat cards row
  const cardW = (CONTENT_WIDTH - 9) / 4;
  addStatCard(doc, PAGE.margin, y, cardW, 22, 'Records Migrated', totalLoaded.toLocaleString());
  addStatCard(doc, PAGE.margin + cardW + 3, y, cardW, 22, 'Success Rate', `${successRate}%`, successRate >= 95 ? COLORS.success : COLORS.warning);
  addStatCard(doc, PAGE.margin + (cardW + 3) * 2, y, cardW, 22, 'AI Fixes Applied', data.aiSuggestionsApplied.toLocaleString());
  addStatCard(doc, PAGE.margin + (cardW + 3) * 3, y, cardW, 22, 'Source Amount', formatCurrency(data.financialReconciliation.sourceTotal));
  y += 30;

  // Entity breakdown table
  y = addSectionHeading(doc, 'Migration Overview', y);
  autoTable(doc, {
    startY: y,
    head: [['Entity Type', 'Source Records', 'Successfully Loaded', 'Failed', 'Success Rate']],
    body: [
      ['Organizations', data.entityCounts.investees.total, data.entityCounts.investees.loaded, data.entityCounts.investees.failed, `${Math.round(data.entityCounts.investees.loaded / Math.max(data.entityCounts.investees.total, 1) * 100)}%`],
      ['Holdings / Funds', data.entityCounts.holdings.total, data.entityCounts.holdings.loaded, data.entityCounts.holdings.failed, `${Math.round(data.entityCounts.holdings.loaded / Math.max(data.entityCounts.holdings.total, 1) * 100)}%`],
      ['Contributions', data.entityCounts.contributions.total, data.entityCounts.contributions.loaded, data.entityCounts.contributions.failed, `${Math.round(data.entityCounts.contributions.loaded / Math.max(data.entityCounts.contributions.total, 1) * 100)}%`],
      ['Impact Metrics', data.entityCounts.metrics.total, data.entityCounts.metrics.loaded, data.entityCounts.metrics.failed, `${Math.round(data.entityCounts.metrics.loaded / Math.max(data.entityCounts.metrics.total, 1) * 100)}%`],
    ],
    theme: 'striped',
    headStyles: { fillColor: COLORS.primary, textColor: COLORS.white, fontSize: 9, fontStyle: 'bold' },
    bodyStyles: { fontSize: 9 },
    columnStyles: { 4: { halign: 'center' } },
    margin: { left: PAGE.margin, right: PAGE.margin },
  });
  y = (doc as any).lastAutoTable.finalY + 8;

  // Financial reconciliation
  y = checkPageBreak(doc, y, 40);
  y = addSectionHeading(doc, 'Financial Reconciliation', y);
  autoTable(doc, {
    startY: y,
    head: [['Metric', 'Value']],
    body: [
      ['Total source contributions amount', formatCurrency(data.financialReconciliation.sourceTotal)],
      ['Total loaded contributions amount', formatCurrency(data.financialReconciliation.loadedTotal)],
      ['Delta', formatCurrency(Math.abs(data.financialReconciliation.sourceTotal - data.financialReconciliation.loadedTotal))],
      ['Delta %', formatPercent(data.financialReconciliation.deltaPercent)],
      ['Reconciliation status', data.financialReconciliation.deltaPercent < 1 ? '✓ Within acceptable tolerance' : '⚠ Review recommended'],
    ],
    theme: 'plain',
    headStyles: { fillColor: COLORS.primaryLight, textColor: COLORS.dark, fontSize: 9 },
    bodyStyles: { fontSize: 9 },
    columnStyles: { 0: { fontStyle: 'bold', cellWidth: 100 } },
    margin: { left: PAGE.margin, right: PAGE.margin },
  });
  y = (doc as any).lastAutoTable.finalY + 8;

  // Data enrichment
  y = checkPageBreak(doc, y, 35);
  y = addSectionHeading(doc, 'AI Data Enrichment', y);
  autoTable(doc, {
    startY: y,
    head: [['Enrichment Type', 'Count']],
    body: [
      ['Charities matched from public database', data.enrichmentStats.charitiesMatched],
      ['Tax years automatically derived', data.enrichmentStats.taxYearsDerived],
      ['Deductible amounts calculated', data.enrichmentStats.deductibleAmountsCalculated],
      ['Data quality issues auto-fixed by AI', data.errorsFixed],
    ],
    theme: 'plain',
    headStyles: { fillColor: COLORS.primaryLight, textColor: COLORS.dark, fontSize: 9 },
    bodyStyles: { fontSize: 9 },
    columnStyles: { 0: { fontStyle: 'bold', cellWidth: 120 } },
    margin: { left: PAGE.margin, right: PAGE.margin },
  });
  y = (doc as any).lastAutoTable.finalY + 8;

  // Action items (if any)
  if (data.actionItems.length > 0) {
    y = checkPageBreak(doc, y, 30);
    y = addSectionHeading(doc, 'Action Items', y);
    doc.setFontSize(9);
    doc.setTextColor(...COLORS.dark);
    for (const item of data.actionItems) {
      y = checkPageBreak(doc, y, 8);
      doc.setFillColor(254, 243, 199); // yellow-100
      doc.roundedRect(PAGE.margin, y - 4, CONTENT_WIDTH, 7, 1, 1, 'F');
      doc.setFont('helvetica', 'bold');

      doc.setFont('helvetica', 'normal');
      doc.text(item, PAGE.margin + 8, y);
      y += 9;

    y += 4;
  }

  // What's new in Benevolence
  y = checkPageBreak(doc, y, 40);
  y = addSectionHeading(doc, "What's New in Benevolence", y);
  const features = [
    'AI-powered impact analytics dashboard with 14 visualization widgets',
    'Real-time portfolio performance tracking with XIRR calculations',
    'Automated tax document processing and IRS compliance checking',
    'Charity ratings integration (Charity Navigator, Candid, ProPublica)',

  doc.setFontSize(9);
  for (const feature of features) {
    y = checkPageBreak(doc, y, 7);
    doc.setTextColor(...COLORS.success);
    doc.text('✓', PAGE.margin + 2, y);
    doc.setTextColor(...COLORS.dark);
    doc.setFont('helvetica', 'normal');
    doc.text(feature, PAGE.margin + 8, y);
    y += 6;
  }

  addPageFooter(doc, `Generated by Benevolence · ${generatedDate} · Confidential`);

  return Buffer.from(doc.output('arraybuffer'));
}
```

```typescript
import { jsPDF } from 'jspdf';
import autoTable from 'jspdf-autotable';
import {
  createDoc, addPageHeader, addPageFooter, addStatCard,
  addSectionHeading, addBodyText, checkPageBreak,
  formatCurrency, formatPercent, COLORS, PAGE, CONTENT_WIDTH
} from './pdf-generator';

export interface BoardReportData {
  portfolioName: string;
  asOfDate: string;
  totalAUM: number;
  totalHoldings: number;
  ytdReturn?: number;
  topHoldings: Array<{
    name: string;
    assetClass: string;
    value: number;
    percentOfPortfolio: number;
    sector?: string;
  }>;
  kpis: Array<{
    name: string;
    value: string;
    unit?: string;
    trend?: 'up' | 'down' | 'flat';
  }>;
  recentGrants: Array<{
    recipient: string;
    amount: number;
    date: string;
    status: string;
  }>;
  impactSummary?: string;
}

export function generateBoardReportPDF(data: BoardReportData): Buffer {
  const doc = createDoc();
  const generatedDate = new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' });

  // Page 1: Portfolio Overview
  let y = addPageHeader(doc, 'Portfolio Report', `${data.portfolioName} · As of ${data.asOfDate}`, 1);

  // Key stats row
  const cw = (CONTENT_WIDTH - 9) / 4;
  addStatCard(doc, PAGE.margin, y, cw, 22, 'Total AUM', formatCurrency(data.totalAUM));
  addStatCard(doc, PAGE.margin + cw + 3, y, cw, 22, 'Holdings', data.totalHoldings.toString());
  addStatCard(doc, PAGE.margin + (cw + 3) * 2, y, cw, 22, 'YTD Return', data.ytdReturn !== undefined ? formatPercent(data.ytdReturn) : 'N/A', data.ytdReturn !== undefined && data.ytdReturn >= 0 ? COLORS.success : COLORS.danger);
  addStatCard(doc, PAGE.margin + (cw + 3) * 3, y, cw, 22, 'Report Date', generatedDate);
  y += 32;

  // Top holdings table
  if (data.topHoldings.length > 0) {
    y = addSectionHeading(doc, 'Top Holdings', y);
    autoTable(doc, {
      startY: y,
      head: [['Holding', 'Asset Class', 'Sector', 'Value', '% of Portfolio']],
      body: data.topHoldings.slice(0, 15).map(h => [
        h.name.length > 35 ? h.name.substring(0, 33) + '…' : h.name,
        h.assetClass ?? '—',
        h.sector ?? '—',
        formatCurrency(h.value),
        formatPercent(h.percentOfPortfolio),
      ]),
      theme: 'striped',
      headStyles: { fillColor: COLORS.primary, textColor: COLORS.white, fontSize: 9, fontStyle: 'bold' },
      bodyStyles: { fontSize: 9 },
      columnStyles: { 3: { halign: 'right' }, 4: { halign: 'right', cellWidth: 28 } },
      margin: { left: PAGE.margin, right: PAGE.margin },
    });
    y = (doc as any).lastAutoTable.finalY + 8;
  }

  // KPIs
  if (data.kpis.length > 0) {
    y = checkPageBreak(doc, y, 40);
    y = addSectionHeading(doc, 'Impact KPIs', y);
    autoTable(doc, {
      startY: y,
      head: [['Metric', 'Value', 'Unit', 'Trend']],
      body: data.kpis.map(k => [
        k.name,
        k.value,
        k.unit ?? '—',
        k.trend === 'up' ? '↑' : k.trend === 'down' ? '↓' : '→',
      ]),
      theme: 'plain',
      headStyles: { fillColor: COLORS.primaryLight, textColor: COLORS.dark, fontSize: 9 },
      bodyStyles: { fontSize: 9 },
      columnStyles: { 3: { halign: 'center', cellWidth: 20 } },
      margin: { left: PAGE.margin, right: PAGE.margin },
    });
    y = (doc as any).lastAutoTable.finalY + 8;
  }

  // Recent grants
  if (data.recentGrants.length > 0) {
    y = checkPageBreak(doc, y, 40);
    y = addSectionHeading(doc, 'Recent Grants', y);
    autoTable(doc, {
      startY: y,
      head: [['Recipient', 'Amount', 'Date', 'Status']],
      body: data.recentGrants.map(g => [
        g.recipient.length > 40 ? g.recipient.substring(0, 38) + '…' : g.recipient,
        formatCurrency(g.amount),
        g.date,
        g.status,
      ]),
      theme: 'striped',
      headStyles: { fillColor: COLORS.primary, textColor: COLORS.white, fontSize: 9 },
      bodyStyles: { fontSize: 9 },
      columnStyles: { 1: { halign: 'right' } },
      margin: { left: PAGE.margin, right: PAGE.margin },
    });
    y = (doc as any).lastAutoTable.finalY + 8;
  }

  // Impact narrative (if provided)
  if (data.impactSummary) {
    y = checkPageBreak(doc, y, 30);
    y = addSectionHeading(doc, 'Impact Summary', y);
    y = addBodyText(doc, data.impactSummary, y);
  }

  addPageFooter(doc, `${data.portfolioName} · Board Report · Confidential`);

  return Buffer.from(doc.output('arraybuffer'));
}
```

```typescript
// GET /api/portfolio/[id]/reports/tax?year=2024
// Generates a PDF tax summary report and returns it as a download
import { NextResponse } from 'next/server';
import { supabasePublic } from '@/lib/supabase';
import { generateTaxReportPDF } from '@/lib/pdf/tax-report';
import { TAX_DISCLAIMER } from '@/lib/tax/constants';
import { CONTRIBUTION_TYPE_LABELS, RECIPIENT_TYPE_LABELS, FILING_STATUS_LABELS } from '@/lib/tax/constants';

export async function GET(req: Request, ctx: { params: Promise<{ id: string }> }) {
  const { id: portfolioId } = await ctx.params;
  const url = new URL(req.url);
  const year = Number(url.searchParams.get('year') || new Date().getFullYear());

  const sb = await supabasePublic();

  const [
    { data: profile },
    { data: contributions },
    { data: carryforwards },
    { data: portfolio },
  ] = await Promise.all([
    sb.from('tax_profiles').select('*').eq('portfolio_id', portfolioId).eq('tax_year', year).single(),
    sb.from('v_tax_contributions_enriched').select('*').eq('portfolio_id', portfolioId).eq('tax_year', year).order('contribution_date', { ascending: true }),
    sb.from('v_active_carryforwards').select('*').eq('portfolio_id', portfolioId).lte('originating_tax_year', year),
    sb.from('portfolios').select('name').eq('id', portfolioId).single(),
  ]);

  if (!portfolio) {
    return NextResponse.json({ error: 'Not found' }, { status: 404 });
  }

  const totalContributions = (contributions || []).reduce((sum, c) => sum + (c.amount_usd || 0), 0);
  const totalDeductible = (contributions || []).reduce((sum, c) => sum + (c.calculated_deductible_amount || c.amount_usd || 0), 0);
  const cashContributions = (contributions || []).filter(c => ['cash','check','wire'].includes(c.contribution_type)).reduce((sum, c) => sum + (c.amount_usd || 0), 0);
  const compliantCount = (contributions || []).filter(c => c.is_compliant).length;

  const pdfBuffer = generateTaxReportPDF({
    portfolioName: portfolio.name,
    taxYear: year,
    filingStatus: profile?.filing_status ? FILING_STATUS_LABELS[profile.filing_status as keyof typeof FILING_STATUS_LABELS] : undefined,
    summary: {
      totalContributions,
      totalDeductible,
      cashContributions,
      nonCashContributions: totalContributions - cashContributions,
      contributionCount: (contributions || []).length,
      complianceRate: (contributions || []).length > 0 ? Math.round((compliantCount / (contributions || []).length) * 100) : null,
      totalCarryforwardAvailable: (carryforwards || []).reduce((sum, c) => sum + (c.amount_remaining || 0), 0),
    },
    contributions: (contributions || []).map(c => ({
      date: c.contribution_date,
      recipient: c.recipient_name,
      recipientEIN: c.recipient_ein || '',
      type: CONTRIBUTION_TYPE_LABELS[c.contribution_type as keyof typeof CONTRIBUTION_TYPE_LABELS] ?? c.contribution_type,
      amount: c.amount_usd,
      deductibleAmount: c.calculated_deductible_amount || c.amount_usd,
      acknowledgmentReceived: c.acknowledgment_received ? 'Yes' : 'No',
      isCompliant: c.is_compliant ? 'Yes' : 'No',
      substantiationRequired: c.substantiation_requirement ?? '',
      notes: c.notes || '',
    })),
    carryforwards: (carryforwards || []).map(cf => ({
      originatingYear: cf.originating_tax_year,
      expiresYear: cf.expires_tax_year,
      category: cf.agi_limit_category,
      originalAmount: cf.amount,
      remainingAmount: cf.amount_remaining,
      recipient: cf.recipient_name || '',
    })),
    disclaimer: TAX_DISCLAIMER,
  });

  return new NextResponse(pdfBuffer, {
    headers: {
      'Content-Type': 'application/pdf',
      'Content-Disposition': `attachment; filename="tax-summary-${year}.pdf"`,
    },
  });
}
```

```typescript
// GET /api/portfolio/[id]/reports/board
// Generates a board-level portfolio PDF report
import { NextResponse } from 'next/server';
import { supabasePublic } from '@/lib/supabase';
import { generateBoardReportPDF } from '@/lib/pdf/board-report';

export async function GET(req: Request, ctx: { params: Promise<{ id: string }> }) {

  const sb = await supabasePublic();

  const [
    { data: portfolio },
    { data: holdings },
    { data: kpis },
    { data: grants },
  ] = await Promise.all([
    sb.from('portfolios').select('*').eq('id', portfolioId).single(),
    sb.from('v_holdings').select('*').eq('portfolio_id', portfolioId).eq('status', 'Active').order('funds_allocated', { ascending: false }).limit(20),
    sb.from('v_portfolio_kpi_latest').select('*').eq('portfolio_id', portfolioId).limit(10),
    sb.from('v_grants').select('*').eq('portfolio_id', portfolioId).order('start_date', { ascending: false }).limit(10),
  ]);

  if (!portfolio) {
    return NextResponse.json({ error: 'Not found' }, { status: 404 });
  }

  const totalAUM = (holdings || []).reduce((sum, h) => sum + (h.funds_allocated || 0), 0);

  const pdfBuffer = generateBoardReportPDF({
    portfolioName: portfolio.name,
    asOfDate: new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }),
    totalAUM,
    totalHoldings: (holdings || []).length,
    topHoldings: (holdings || []).map(h => ({
      name: h.name,
      assetClass: h.asset_class ?? 'Other',
      value: h.funds_allocated ?? 0,
      percentOfPortfolio: totalAUM > 0 ? ((h.funds_allocated ?? 0) / totalAUM) * 100 : 0,
      sector: h.sector ?? undefined,
    })),
    kpis: (kpis || []).map(k => ({
      name: k.display_name ?? k.metric_code,
      value: k.latest_value?.toString() ?? 'N/A',
      unit: k.unit ?? undefined,
    })),
    recentGrants: (grants || []).map(g => ({
      recipient: g.investee_name ?? 'Unknown',
      amount: g.amount_committed ?? 0,
      date: g.start_date ?? '',
      status: g.status ?? 'Unknown',
    })),
  });

  return new NextResponse(pdfBuffer, {
    headers: {
      'Content-Type': 'application/pdf',
      'Content-Disposition': `attachment; filename="board-report-${new Date().toISOString().split('T')[0]}.pdf"`,
    },
  });
}
```

```typescript
// GET /api/admin/imports/:id/report?format=pdf|markdown
// If format=pdf, generate PDF using jsPDF and return as download
// If format=markdown (default), return existing markdown behavior
// Gather stats from import_jobs + staging counts + reconciliation_data
// Call generateMigrationReportPDF() and return buffer

import { NextResponse } from 'next/server';
import { createAdminClient } from '@/lib/supabase';
import { generateMigrationReportPDF } from '@/lib/pdf/migration-report';

export async function GET(req: Request, ctx: { params: Promise<{ id: string }> }) {
  const { id: importJobId } = await ctx.params;
  const url = new URL(req.url);
  const format = url.searchParams.get('format') ?? 'markdown';

  const supabase = createAdminClient();
  const { data: job } = await supabase.from('import_jobs').select('*').eq('id', importJobId).single();

  if (!job) return NextResponse.json({ error: 'Not found' }, { status: 404 });

  if (format !== 'pdf') {
    // Return existing markdown behavior or simple JSON
    return NextResponse.json({ data: job.reconciliation_data });
  }

  // Gather entity counts from staging tables
  const entities = ['investees', 'holdings', 'contributions', 'metrics'] as const;
  const entityCounts: Record<string, { loaded: number; failed: number; total: number }> = {};
  for (const entity of entities) {
    const table = `staging_import_${entity}`;
    const { count: total } = await supabase.from(table).select('*', { count: 'exact', head: true }).eq('import_job_id', importJobId);
    const { count: loaded } = await supabase.from(table).select('*', { count: 'exact', head: true }).eq('import_job_id', importJobId).in('action_taken', ['create', 'update']);
    const { count: failed } = await supabase.from(table).select('*', { count: 'exact', head: true }).eq('import_job_id', importJobId).eq('action_taken', 'error');
    entityCounts[entity] = { total: total ?? 0, loaded: loaded ?? 0, failed: failed ?? 0 };
  }

  // AI suggestions applied
  const { count: aiApplied } = await supabase.from('import_ai_suggestions').select('*', { count: 'exact', head: true }).eq('import_job_id', importJobId).in('status', ['accepted', 'auto_applied']);

  const reconciliation = job.reconciliation_data as Record<string, unknown> | null;

  const pdfBuffer = generateMigrationReportPDF({
    jobName: job.name,
    portfolioName: job.portfolio_id ?? 'Unknown Portfolio',
    sourceSystem: job.source_type === 'blackbaud_api' ? 'Blackbaud RE NXT' : job.source_type === 'csv_export' ? 'CSV Export' : 'Direct Database',
    completedAt: job.completed_at ?? new Date().toISOString(),
    entityCounts: entityCounts as any,
    financialReconciliation: {
      sourceTotal: (reconciliation?.contributions as any)?.sourceTotalAmount ?? 0,
      loadedTotal: (reconciliation?.contributions as any)?.loadedTotalAmount ?? 0,
      deltaPercent: (reconciliation?.contributions as any)?.amountDeltaPercent ?? 0,
    },
    errorsFixed: job.records_failed ?? 0,
    aiSuggestionsApplied: aiApplied ?? 0,
    actionItems: (reconciliation?.actionItems as string[]) ?? [],
    enrichmentStats: {
      charitiesMatched: (reconciliation?.enrichment as any)?.charitiesMatched ?? 0,
      taxYearsDerived: (reconciliation?.enrichment as any)?.taxYearsDerived ?? 0,
      deductibleAmountsCalculated: (reconciliation?.enrichment as any)?.deductibleAmountsCalculated ?? 0,
    },
  });

  return new NextResponse(pdfBuffer, {
    headers: {
      'Content-Type': 'application/pdf',
      'Content-Disposition': `attachment; filename="migration-report-${importJobId.slice(0, 8)}.pdf"`,
    },
  });
}
```
